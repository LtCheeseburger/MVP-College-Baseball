cmake_minimum_required(VERSION 3.25)
project(uniform_tests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(FetchContent)
# Same revision as sdk/cmake/XbeRecomp.cmake. These tests do not build or distribute generated game code/assets.
FetchContent_Declare(tomlplusplus GIT_REPOSITORY https://github.com/marzer/tomlplusplus.git GIT_TAG v3.4.0)
FetchContent_MakeAvailable(tomlplusplus)
set(ROOT ${CMAKE_CURRENT_LIST_DIR}/../..)
enable_testing()
add_executable(uniform_tests uniforms_test.cpp ${ROOT}/src/common/uniforms.cpp
    ${ROOT}/src/common/uniform_catalog.cpp ${ROOT}/src/common/uniform_layout.cpp)
target_include_directories(uniform_tests PRIVATE ${ROOT}/src)
target_link_libraries(uniform_tests PRIVATE tomlplusplus::tomlplusplus)
add_test(NAME uniform_selections_catalog_layout COMMAND uniform_tests)
add_executable(uniform_hooks_test hooks_test.cpp ${ROOT}/src/common/uniforms.cpp ${ROOT}/src/game/uniform_components.cpp)
target_include_directories(uniform_hooks_test PRIVATE ${ROOT}/src ${ROOT}/sdk/runtime)
add_test(NAME uniform_guest_boundary COMMAND uniform_hooks_test)
find_package(Python3 COMPONENTS Interpreter)
if(Python3_FOUND)
    add_test(NAME uniform_piece_authoring COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_LIST_DIR}/test_authoring.py)
endif()
